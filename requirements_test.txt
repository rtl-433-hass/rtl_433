# Testing dependencies for the rtl_433 integration.
#
# pytest-homeassistant-custom-component pulls in the matching Home Assistant
# version together with the core pytest stack it ships, including:
#   - pytest
#   - pytest-asyncio
#   - pytest-aiohttp
#   - pytest-cov
#   - pytest-timeout
#   - pytest-xdist (parallel test execution)
#   - coverage
#   - freezegun (time mocking)
#
# Install with: uv pip install -r requirements_test.txt

# Testing utilities for Home Assistant custom components.
# Provides additional fixtures and utilities specifically for custom-component
# testing, and pins the Home Assistant version used for the test run.
# Requires Python 3.14 (0.13.317+) and pulls in Home Assistant 2026.4+,
# matching the minimum versions declared in pyproject.toml and hacs.json.
pytest-homeassistant-custom-component==0.13.362

# The integration's runtime dependency (declared in manifest.json /
# requirements.txt). Installed here so the pytest environment can import the
# rtl_433 protocol layer under test.
pyrtl_433==0.4.0

# The Home Assistant frontend bundle. Not used by any test directly, but the
# integration declares ``panel_custom`` (and through it ``frontend``) as a
# manifest dependency, and Home Assistant refuses to set up a component whose
# dependencies fail -- ``frontend`` fails to import ``hass_frontend`` without
# this package, which would take every hub-setup test down with it.
# pytest-homeassistant-custom-component does not pull it in, so it is pinned here
# to the version the installed Home Assistant's frontend manifest requires.
home-assistant-frontend==20260826.4

# Mutation testing. mutmut 3.x copies the package into a mutants/ working tree
# and forks once per mutant, running the pytest suite with coverage-based test
# selection. Configured under [tool.mutmut] in pyproject.toml; the CI mutation
# workflow and mutmut-ratchet (below) depend on it.
mutmut==3.7.0

# The mutation CI gate: PR scoping, time-balanced sharding, per-file stats, and
# the score ratchet. Shared with the pyrtl_433 repository, which is why it is a
# package rather than the scripts/mutation_*.py helpers it replaced. Configured
# under [tool.mutmut_ratchet] in pyproject.toml.
# https://github.com/rtl-433-hass/mutmut-ratchet
mutmut-ratchet>=0.1.0
